189    def calculate_days_until_game(self, game_date_str):
190        """
191        Calculate the integer difference between today and the scheduled game.
192
193        Args:
194            game_date_str (str): Date in 'YYYY-MM-DD' format.
195
196        Returns:
197            int or str: Number of days remaining, or empty string if invalid.
198        """
199        if not game_date_str:
200            return ""
201
202        try:
203            game_date = datetime.strptime(game_date_str, "%Y-%m-%d").date()
204        except ValueError:
205            return ""
206
207        today = date.today()
208        date_difference = (game_date - today).days
209
210        return date_difference if date_difference >= 0 else ""

401        """
